use regex::Regex;
use rusqlite::{params, Connection, Result};
use scraper::{Html, Selector};
use serde::Serialize;
use std::collections::HashMap;

#[derive(Clone, Debug, Serialize, Default)]
pub struct SearchResult {
    pub title: String,
    pub url: String,
    pub snippet: String,
    pub score: f64,
}

pub struct SearchIndex {
    conn: Connection,
}

impl SearchIndex {
    pub fn open(path: &str) -> Result<Self> {
        let conn = Connection::open(path)?;
        conn.execute_batch(
            r#"
            CREATE TABLE IF NOT EXISTS documents (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                url TEXT UNIQUE NOT NULL,
                title TEXT NOT NULL,
                body TEXT NOT NULL
            );

            CREATE TABLE IF NOT EXISTS term_counts (
                doc_id INTEGER NOT NULL,
                term TEXT NOT NULL,
                count INTEGER NOT NULL,
                PRIMARY KEY(doc_id, term)
            );

            CREATE INDEX IF NOT EXISTS idx_term_counts_term ON term_counts(term);
            "#,
        )?;
        Ok(Self { conn })
    }

    pub fn index_seed_file(&self, seed_file: &str) -> Result<()> {
        let contents = std::fs::read_to_string(seed_file).unwrap_or_else(|_| {
            eprintln!("Warning: {} not found, using built-in defaults.", seed_file);
            "https://www.rust-lang.org/\nhttps://www.wikipedia.org/\nhttps://www.example.com/\n".to_string()
        });

        for line in contents.lines() {
            let url = line.trim();
            if !url.is_empty() {
                let _ = self.index_url(url);
            }
        }
        Ok(())
    }

    pub fn search(&self, query: &str, limit: usize) -> Result<Vec<SearchResult>> {
        let terms = Self::tokenize(query);
        if terms.is_empty() {
            return Ok(vec![]);
        }

        let doc_count: f64 = self
            .conn
            .query_row("SELECT COUNT(*) FROM documents", [], |row| row.get(0))? as f64;

        let mut scores: HashMap<i64, f64> = HashMap::new();
        let mut doc_ids: Vec<i64> = vec![];

        for term in &terms {
            let df: i64 = self.conn.query_row(
                "SELECT COUNT(DISTINCT doc_id) FROM term_counts WHERE term = ?",
                [term],
                |row| row.get(0),
            )?;

            let mut stmt = self.conn.prepare(
                "SELECT doc_id, count FROM term_counts WHERE term = ?",
            )?;
            let rows = stmt.query_map([term], |row| Ok((row.get::<_, i64>(0)?, row.get::<_, i64>(1)?)))?;

            for row in rows {
                let (doc_id, count) = row?;
                let idf = ((doc_count + 1.0) / (df as f64 + 1.0)).ln() + 1.0;
                let score = count as f64 * idf;
                *scores.entry(doc_id).or_insert(0.0) += score;
                if !doc_ids.contains(&doc_id) {
                    doc_ids.push(doc_id);
                }
            }
        }

        let mut results: Vec<SearchResult> = Vec::new();
        for doc_id in doc_ids {
            let (url, title, body): (String, String, String) = self.conn.query_row(
                "SELECT url, title, body FROM documents WHERE id = ?",
                [doc_id],
                |row| Ok((row.get(0)?, row.get(1)?, row.get(2)?)),
            )?;

            let title_lower = title.to_lowercase();
            let title_bonus = terms
                .iter()
                .filter(|term| title_lower.contains(term))
                .count() as f64
                * 2.0;

            let score = scores.get(&doc_id).copied().unwrap_or(0.0) + title_bonus;
            let snippet = snippet_for(&body, &terms);

            results.push(SearchResult {
                title,
                url,
                snippet,
                score,
            });
        }

        results.sort_by(|a, b| b.score.partial_cmp(&a.score).unwrap_or(std::cmp::Ordering::Equal));
        results.truncate(limit);
        Ok(results)
    }

    fn tokenize(text: &str) -> Vec<String> {
        let re = Regex::new(r"[A-Za-z0-9]+").expect("tokenizer regex is valid");
        re.find_iter(text)
            .map(|m| m.as_str().to_lowercase())
            .filter(|part| !part.is_empty())
            .collect()
    }

    fn index_url(&self, raw_url: &str) -> Result<()> {
        let url = raw_url.trim();
        if url.is_empty() {
            return Ok(());
        }

        let response = reqwest::blocking::get(url);
        let html = match response {
            Ok(resp) => match resp.text() {
                Ok(body) => body,
                Err(_) => return Ok(()),
            },
            Err(_) => return Ok(()),
        };

        let title = extract_title(&html).unwrap_or_else(|| url.to_string());
        let body = extract_main_text(&html);
        let text_for_index = format!("{} {}", title, body);
        let terms = Self::tokenize(&text_for_index);
        if terms.is_empty() {
            return Ok(());
        }

        let inserted = self.conn.execute(
            "INSERT OR IGNORE INTO documents (url, title, body) VALUES (?, ?, ?)",
            params![url, title, body],
        )?;
        if inserted == 0 {
            return Ok(());
        }

        let doc_id: i64 = self.conn.query_row(
            "SELECT id FROM documents WHERE url = ?",
            [url],
            |row| row.get(0),
        )?;

        let mut counts: HashMap<String, i32> = HashMap::new();
        for term in terms {
            *counts.entry(term).or_insert(0) += 1;
        }

        for (term, count) in counts {
            self.conn.execute(
                "INSERT INTO term_counts (doc_id, term, count) VALUES (?, ?, ?)",
                params![doc_id, term, count],
            )?;
        }

        Ok(())
    }
}

fn extract_title(html: &str) -> Option<String> {
    let doc = Html::parse_document(html);
    let selector = Selector::parse("title").ok()?;
    doc.select(&selector)
        .next()
        .map(|node| node.text().collect::<String>().trim().to_string())
}

fn extract_main_text(html: &str) -> String {
    let doc = Html::parse_document(html);
    let selector = Selector::parse("body").unwrap();
    let body_text = match doc.select(&selector).next() {
        Some(node) => node.text().collect::<String>(),
        None => doc.root().text().collect::<String>(),
    };
    let re = Regex::new(r"\s+").expect("whitespace regex is valid");
    re.replace_all(&body_text, " ").trim().to_string()
}

fn snippet_for(body: &str, terms: &[String]) -> String {
    let words: Vec<&str> = body.split_whitespace().collect();
    if words.is_empty() {
        return String::new();
    }

    let mut best_start: usize = 0;
    let mut best_score: i32 = i32::MIN;

    for start in 0..words.len().saturating_sub(1).min(40) {
        let mut score = 0;
        for term in terms {
            let target = term.as_str();
            for i in start..(start + 25).min(words.len()) {
                if words[i].to_lowercase().contains(target) {
                    score += 1;
                }
            }
        }
        if score > best_score {
            best_score = score;
            best_start = start;
        }
    }

    let end = (best_start + 30).min(words.len());
    let snippet: String = words[best_start..end].join(" ");
    if snippet.len() > 220 {
        format!("{}...", &snippet[..217])
    } else {
        snippet
    }
}
